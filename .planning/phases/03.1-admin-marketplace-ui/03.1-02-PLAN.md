import type { HttpClient } from '../../services/http/client';
import type {

export type AdminApi = {

  // Reviews
  listReviews: (params?: { status?: string; limit?: number; offset?: number }) => Promise<PaginatedResponse<PublishedAgent>>;
  approveAgent: (id: string) => Promise<void>;
  rejectAgent: (id: string, reason: string) => Promise<void>;

export function createAdminApi(client: HttpClient): AdminApi {
  const apiPrefix = '/api/v1/admin';

  return {
    getStats: () => client.get<AdminStats>(`${apiPrefix}/stats`),

    listChannels: (params) => {
      const qs = new URLSearchParams();
      if (params) Object.entries(params).forEach(([k, v]) => { if (v !== undefined) qs.set(k, String(v)); });
      return client.get<PaginatedResponse<ChannelInfo>>(`${apiPrefix}/channels?${qs.toString()}`);
    },
    createChannel: (input) => client.post<ChannelInfo>(`${apiPrefix}/channels`, input),
    updateChannel: (id, input) => client.put<ChannelInfo>(`${apiPrefix}/channels/${id}`, input),
    deleteChannel: (id) => client.delete<void>(`${apiPrefix}/channels/${id}`),
    testChannel: (id) => client.post<ChannelTestResult>(`${apiPrefix}/channels/${id}/test`),
    getChannelHealth: (id) => client.get<{ status: string; latency: number }>(`${apiPrefix}/channels/${id}/health`),
    batchUpdateChannels: (ids, enabled) => client.post<void>(`${apiPrefix}/channels/batch`, { ids, enabled }),

    listRoutes: () => client.get<RouteInfo[]>(`${apiPrefix}/routes`),
    createRoute: (input) => client.post<RouteInfo>(`${apiPrefix}/routes`, input),
    updateRoute: (id, input) => client.put<RouteInfo>(`${apiPrefix}/routes/${id}`, input),
    deleteRoute: (id) => client.delete<void>(`${apiPrefix}/routes/${id}`),

    listPlans: (params) => {
      const qs = new URLSearchParams();
      if (params) Object.entries(params).forEach(([k, v]) => { if (v !== undefined) qs.set(k, String(v)); });
      return client.get<PlanInfo[]>(`${apiPrefix}/plans?${qs.toString()}`);
    },
    createPlan: (input) => client.post<PlanInfo>(`${apiPrefix}/plans`, input),
    updatePlan: (id, input) => client.put<PlanInfo>(`${apiPrefix}/plans/${id}`, input),
    deactivatePlan: (id) => client.delete<void>(`${apiPrefix}/plans/${id}`),

    listUsers: (params) => {
      const qs = new URLSearchParams();
      if (params) Object.entries(params).forEach(([k, v]) => { if (v !== undefined) qs.set(k, String(v)); });
      return client.get<PaginatedResponse<UserDetail>>(`${apiPrefix}/users?${qs.toString()}`);
    },
    getUser: (id) => client.get<UserDetail>(`${apiPrefix}/users/${id}`),
    updateUser: (id, input) => client.put<UserDetail>(`${apiPrefix}/users/${id}`, input),
    disableUser: (id) => client.post<void>(`${apiPrefix}/users/${id}/disable`),
    enableUser: (id) => client.post<void>(`${apiPrefix}/users/${id}/enable`),

    listAuditLogs: (params) => {
      const qs = new URLSearchParams();
      if (params) Object.entries(params).forEach(([k, v]) => { if (v !== undefined) qs.set(k, String(v)); });
      return client.get<PaginatedResponse<AuditEntry>>(`${apiPrefix}/audit-logs?${qs.toString()}`);
    },

    listReviews: (params) => {
      const qs = new URLSearchParams();
      if (params) Object.entries(params).forEach(([k, v]) => { if (v !== undefined) qs.set(k, String(v)); });
      return client.get<PaginatedResponse<PublishedAgent>>(`${apiPrefix}/reviews?${qs.toString()}`);
    },
    approveAgent: (id) => client.post<void>(`${apiPrefix}/reviews/${id}/approve`),
    rejectAgent: (id, reason) => client.post<void>(`${apiPrefix}/reviews/${id}/reject`, { reason }),
  };
}

export type Category = { id: string; name: string; slug: string; agentCount: number };

export type MarketplaceHomeData = {

export type MarketplaceApi = {
  // Homepage
  getFeatured: () => Promise<MarketplaceAgent[]>;
  getCuratedSections: () => Promise<{ popular: MarketplaceAgent[]; topRated: MarketplaceAgent[]; recent: MarketplaceAgent[] }>;
  getCategories: () => Promise<Category[]>;
  // Agent CRUD
  getAgent: (id: string) => Promise<MarketplaceAgent>;
  publishAgent: (input: AgentPublishRequest) => Promise<MarketplaceAgent>;
  updateAgent: (id: string, input: Partial<AgentPublishRequest>) => Promise<MarketplaceAgent>;
  // Search
  searchAgents: (params: MarketplaceSearchParams) => Promise<{ agents: MarketplaceAgent[]; total: number }>;
  // Install
  installAgent: (agentId: string, versionId?: string) => Promise<AgentInstall>;
  uninstallAgent: (installId: string) => Promise<void>;
  getInstalledAgents: () => Promise<AgentInstall[]>;
  // My Agents
  getMyAgents: (limit?: number, offset?: number) => Promise<MarketplaceAgent[]>;
  // Reviews
  getReviews: (agentId: string, limit?: number, offset?: number) => Promise<AgentReview[]>;
  submitReview: (agentId: string, input: { rating: number; text: string }) => Promise<AgentReview>;
  // Versions
  getVersions: (agentId: string) => Promise<AgentVersion[]>;
};

export function createMarketplaceApi(client: HttpClient): MarketplaceApi {
  const apiPrefix = '/api/v1/marketplace';

  return {
    getFeatured: () => client.get<MarketplaceAgent[]>(`${apiPrefix}/featured`),
    getCuratedSections: () => client.get<{ popular: MarketplaceAgent[]; topRated: MarketplaceAgent[]; recent: MarketplaceAgent[] }>(`${apiPrefix}/curated`),
    getCategories: () => client.get<Category[]>(`${apiPrefix}/categories`),

    getAgent: (id) => client.get<MarketplaceAgent>(`${apiPrefix}/agents/${id}`),
    publishAgent: (input) => client.post<MarketplaceAgent>(`${apiPrefix}/agents`, input),
    updateAgent: (id, input) => client.put<MarketplaceAgent>(`${apiPrefix}/agents/${id}`, input),

    searchAgents: (params) => {
      const qs = new URLSearchParams();
      Object.entries(params).forEach(([k, v]) => {
        if (v !== undefined && v !== '') qs.set(k, String(v));
      });
      return client.get<{ agents: MarketplaceAgent[]; total: number }>(`${apiPrefix}/search?${qs.toString()}`);
    },

    installAgent: (agentId, versionId) => {
      const qs = versionId ? `?versionId=${versionId}` : '';
      return client.post<AgentInstall>(`${apiPrefix}/agents/${agentId}/install${qs}`);
    },
    uninstallAgent: (installId) => client.delete<void>(`${apiPrefix}/installs/${installId}`),
    getInstalledAgents: () => client.get<AgentInstall[]>(`${apiPrefix}/installs`),
